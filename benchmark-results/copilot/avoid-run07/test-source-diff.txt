+test('POST /api/todos/clear-completed with confirm:true deletes completed todos', async () => {
+  // Create one completed and one incomplete todo
+  const r1 = await fetch(`${BASE_URL}/api/todos`, {
+    method: 'POST', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ title: 'completed-task' })
+  });
+  const t1 = await r1.json();
+  await fetch(`${BASE_URL}/api/todos/${t1.id}`, {
+    method: 'PUT', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ completed: true })
+  });
+  const r2 = await fetch(`${BASE_URL}/api/todos`, {
+    method: 'POST', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ title: 'incomplete-task' })
+  });
+  const t2 = await r2.json();
+
+  const beforeAll = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  const completedCount = beforeAll.filter(t => t.completed).length;
+  const incompleteCount = beforeAll.filter(t => !t.completed).length;
+
+  const res = await fetch(`${BASE_URL}/api/todos/clear-completed`, {
+    method: 'POST', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ confirm: true })
+  });
+  assert.strictEqual(res.status, 200);
+  const data = await res.json();
+  assert.strictEqual(typeof data.deleted, 'number');
+  assert.strictEqual(typeof data.remaining, 'number');
+  assert.strictEqual(data.deleted, completedCount);
+  assert.strictEqual(data.remaining, incompleteCount);
+
+  const afterAll = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  assert.ok(afterAll.every(t => !t.completed), 'No completed todos should remain');
+  assert.ok(afterAll.some(t => t.id === t2.id), 'Incomplete todo should still exist');
+});
