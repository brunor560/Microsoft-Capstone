+test('POST /api/todos/clear-completed with confirm:true deletes completed todos', async () => {
+  // Ensure at least one completed and one incomplete todo exist
+  const c1 = await (await fetch(`${BASE_URL}/api/todos`, {
+    method: 'POST',
+    headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ title: 'clear-completed completed item' })
+  })).json();
+  await fetch(`${BASE_URL}/api/todos/${c1.id}`, {
+    method: 'PUT',
+    headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ completed: true })
+  });
+  const c2 = await (await fetch(`${BASE_URL}/api/todos`, {
+    method: 'POST',
+    headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ title: 'clear-completed incomplete item' })
+  })).json();
+
+  const listBefore = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  const completedCount = listBefore.filter(t => t.completed).length;
+  const incompleteCount = listBefore.filter(t => !t.completed).length;
+
+  const res = await fetch(`${BASE_URL}/api/todos/clear-completed`, {
+    method: 'POST',
+    headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ confirm: true })
+  });
+  assert.strictEqual(res.status, 200);
+  const data = await res.json();
+  assert.strictEqual(data.deleted, completedCount);
+  assert.strictEqual(data.remaining, incompleteCount);
+
+  const listAfter = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  assert.strictEqual(listAfter.some(t => t.completed), false);
+  assert.ok(listAfter.some(t => t.id === c2.id));
+});
