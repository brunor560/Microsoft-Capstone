+test('POST /api/todos/clear-completed with confirm:true deletes completed todos', async () => {
+  // Ensure at least one completed and one incomplete todo exist
+  const c1 = await (await fetch(`${BASE_URL}/api/todos`, {
+    method: 'POST', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ title: 'completed-to-clear' })
+  })).json();
+  await fetch(`${BASE_URL}/api/todos/${c1.id}`, {
+    method: 'PUT', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ completed: true })
+  });
+  const c2 = await (await fetch(`${BASE_URL}/api/todos`, {
+    method: 'POST', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ title: 'incomplete-to-keep' })
+  })).json();
+
+  const listBefore = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  const completedCount = listBefore.filter(t => t.completed).length;
+
+  const res = await fetch(`${BASE_URL}/api/todos/clear-completed`, {
+    method: 'POST', headers: { 'Content-Type': 'application/json' },
+    body: JSON.stringify({ confirm: true })
+  });
+  assert.strictEqual(res.status, 200);
+  const data = await res.json();
+  assert.strictEqual(data.deleted, completedCount);
+  assert.ok(typeof data.remaining === 'number');
+
+  const listAfter = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  assert.ok(listAfter.every(t => !t.completed));
+  assert.ok(listAfter.some(t => t.id === c2.id));
+});

+test('POST /api/todos/clear-completed without confirm:true returns 400 and deletes nothing', async () => {

+
+  const cases = [

+    JSON.stringify({ confirm: false }),
+    JSON.stringify({ confirm: 'yes' })
+  ];
+
+  for (const body of cases) {
+    const opts = body
+      ? { method: 'POST', headers: { 'Content-Type': 'application/json' }, body }
+      : { method: 'POST' };
+    const res = await fetch(`${BASE_URL}/api/todos/clear-completed`, opts);
+    assert.strictEqual(res.status, 400);
+    const data = await res.json();
+    assert.ok(typeof data.error === 'string');
+  }
+
+  const listAfter = await (await fetch(`${BASE_URL}/api/todos`)).json();
+  assert.deepStrictEqual(listAfter, listBefore);
+});
