    // Per-todo editor state: id -> { draft, error, saving }
    const editing = new Map();
    let focusRequest = null; // { id, target: 'input' | 'edit' }
    let currentTodos = [];

    function el(tag, props, children) {
      const node = document.createElement(tag);
      Object.assign(node, props || {});
      (children || []).forEach(c => node.append(c));
      return node;
    }

    function renderItem(item) {
      const li = el('li');
      li.dataset.todoId = String(item.id);
      const state = editing.get(item.id);

      if (state) {
        const input = el('input', { type: 'text', value: state.draft, className: 'edit-input' });
        input.setAttribute('aria-label', 'Edit title');
        input.addEventListener('input', () => { state.draft = input.value; });
        input.addEventListener('keydown', (e) => {
          if (e.key === 'Enter') { e.preventDefault(); saveEdit(item.id); }
          else if (e.key === 'Escape') { e.preventDefault(); cancelEdit(item.id); }
        });
        const save = el('button', { type: 'button', textContent: 'Save', className: 'save-btn' });
        save.addEventListener('click', () => saveEdit(item.id));
        const cancel = el('button', { type: 'button', textContent: 'Cancel', className: 'cancel-btn' });
        cancel.addEventListener('click', () => cancelEdit(item.id));
        li.append(input, save, cancel);
        if (state.error) {
          const alertEl = el('div', { className: 'error', textContent: state.error });
          alertEl.setAttribute('role', 'alert');
          li.append(alertEl);
        }
        if (focusRequest && focusRequest.id === item.id && focusRequest.target === 'input') {
          setTimeout(() => input.focus(), 0);
          focusRequest = null;
        }
      } else {
        const span = el('span', { className: 'item-text' + (item.completed ? ' done' : ''), textContent: item.title });
        span.addEventListener('click', () => toggleStatus(item.id, !item.completed));
        const edit = el('button', { type: 'button', textContent: 'Edit', className: 'edit-btn' });
        edit.addEventListener('click', () => startEdit(item.id, item.title));
        const del = el('button', { type: 'button', textContent: 'Delete', className: 'delete-btn' });
        del.addEventListener('click', () => deleteItem(item.id));
        li.append(span, edit, del);
        if (focusRequest && focusRequest.id === item.id && focusRequest.target === 'edit') {
          setTimeout(() => edit.focus(), 0);
          focusRequest = null;
        }
      }
      return li;
    }

    function render() {
      list.replaceChildren(...currentTodos.map(renderItem));
    }

    async function fetchTodos() {
      const res = await fetch('/api/todos');
      currentTodos = await res.json();
      for (const id of [...editing.keys()]) {
        if (!currentTodos.some(t => t.id === id)) editing.delete(id);
      }
      render();
    }

    function startEdit(id, title) {
      editing.set(id, { draft: title, error: '' });
      focusRequest = { id, target: 'input' };
      render();
    }

    function cancelEdit(id) {
      editing.delete(id);
      focusRequest = { id, target: 'edit' };
      render();
    }

    async function saveEdit(id) {
      const state = editing.get(id);
      if (!state || state.saving) return;
      const title = state.draft.trim();
      if (title === '') {
        state.error = 'Title cannot be empty.';
        focusRequest = { id, target: 'input' };
        render();
        return;
      }
      state.saving = true;
      try {
        const res = await fetch(`/api/todos/${id}`, {
          method: 'PUT',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ title })
        });
        if (!res.ok) throw new Error(`Server responded with status ${res.status}`);
        const updated = await res.json();
        const idx = currentTodos.findIndex(t => t.id === id);
        if (idx !== -1) currentTodos[idx] = updated;
        editing.delete(id);
        focusRequest = { id, target: 'edit' };
      } catch (err) {
        state.error = `Could not save title: ${err.message}`;
        focusRequest = { id, target: 'input' };
      } finally {
        state.saving = false;
      }
      render();
    }
